\section{Algorithme}

\subsection{Point le Plus Proche (PPP)}

\begin{tcolorbox}[colback=darkWhite!75!white,colframe=black!30,left=5pt,right=5pt,top=5pt,bottom=5pt]
\textbf{Algorithme : Procédure PPP (Point le Plus Proche)} \\
\textbf{Description :} Construit un cycle hamiltonien en insérant itérativement le point le plus proche du cycle actuel. \\
\textbf{Vars :}
\begin{itemize}
    \item $c$ : Liste d'entiers (Cycle hamiltonien)
    \item $l$ : Réel (Longueur du cycle)
    \item $Q_i, Q_j$ : Entiers (Indices des points sélectionnés)
\end{itemize}
\textbf{Entrées :}
\begin{itemize}
    \item $L$ : Liste de points $P_1, \dots, P_n$
    \item $Q$ : Un point de départ choisi dans $L$
    \item $D$ : La matrice des distances $n \times n$
\end{itemize}
\textbf{Sorties :}
\begin{itemize}
    \item $c$ : Un cycle hamiltonien
    \item $l$ : La longueur du cycle
\end{itemize}
\textbf{Début :}
\begin{enumerate}
    \item $c \leftarrow$ cycle trivial composé du seul point $Q$
    \item $l \leftarrow 0$
    \item Retirer $Q$ de $L$
    \item Tant que $L$ n'est pas vide faire :
    \item \quad $(Q_i, Q_j) \leftarrow \texttt{TrouverCoupleMin}(L, c, D)$
    \item \quad $(c, l) \leftarrow \texttt{InsérerDansCycle}(c, Q_i, Q_j, D, l)$
    \item \quad Retirer $Q_i$ de $L$
    \item Retourner $c, l$
\end{enumerate}
\end{tcolorbox}

\vspace{5mm}

\begin{tcolorbox}[colback=darkWhite!75!white,colframe=black!30,left=5pt,right=5pt,top=5pt,bottom=5pt]
\textbf{Algorithme : InsérerDansCycle} \\
\textbf{Description :} Insère $Q_i$ dans $c$ soit entre $Q_j$ et son successeur, soit entre son prédécesseur et $Q_j$, selon le coût minimal, et met à jour la longueur du cycle. \\
\textbf{Vars :}
\begin{itemize}
    \item $Q_{next}, Q_{prev}$ : Entiers (Indices des voisins de $Q_j$)
    \item $\text{coût}_{\text{droite}}, \text{coût}_{\text{gauche}}$ : Réels (Coûts d'insertion)
\end{itemize}
\textbf{Entrées :}
\begin{itemize}
    \item $c$ : Cycle partiel
    \item $Q_i$ : Point à insérer
    \item $Q_j$ : Point d'ancrage dans $c$
    \item $D$ : Matrice des distances
    \item $l$ : Longueur actuelle du cycle
\end{itemize}
\textbf{Sorties :} $c$ (cycle mis à jour), $l$ (nouvelle longueur) \\
\textbf{Début :}
\begin{enumerate}
    \item Soit $Q_{next}$ le successeur de $Q_j$ dans $c$
    \item Soit $Q_{prev}$ le prédécesseur de $Q_j$ dans $c$
    \item $\text{coût}_{\text{droite}} \leftarrow D[Q_j, Q_i] + D[Q_i, Q_{next}] - D[Q_j, Q_{next}]$
    \item $\text{coût}_{\text{gauche}} \leftarrow D[Q_{prev}, Q_i] + D[Q_i, Q_j] - D[Q_{prev}, Q_j]$
    \item Si $\text{coût}_{\text{droite}} \leq \text{coût}_{\text{gauche}}$ alors :
    \item \quad Insérer $Q_i$ entre $Q_j$ et $Q_{next}$ dans $c$
    \item \quad $l \leftarrow l + \text{coût}_{\text{droite}}$
    \item Sinon :
    \item \quad Insérer $Q_i$ entre $Q_{prev}$ et $Q_j$ dans $c$
    \item \quad $l \leftarrow l + \text{coût}_{\text{gauche}}$
    \item Retourner $c, l$
\end{enumerate}
\end{tcolorbox}


\begin{tcolorbox}[colback=darkWhite!75!white,colframe=black!30,left=5pt,right=5pt,top=5pt,bottom=5pt]
\textbf{Algorithme : TrouverCoupleMin} \\
\textbf{Description :} Trouve le couple $(Q_i, Q_j)$ tel que $Q_i \in L$, $Q_j \in c$ et la distance $D[Q_i, Q_j]$ est minimale. \\
\textbf{Vars :}
\begin{itemize}
    \item $d_{min}$ : Réel (Distance minimale trouvée)
    \item $u, v$ : Entiers (Indices de parcours)
\end{itemize}
\textbf{Entrées :}
\begin{itemize}
    \item $L$ : Liste des points non encore visités
    \item $c$ : Le cycle partiel actuel
    \item $D$ : La matrice des distances $n \times n$
\end{itemize}
\textbf{Sorties :} $Q_i$ (point sélectionné dans $L$), $Q_j$ (point d'ancrage dans $c$) \\
\textbf{Début :}
\begin{enumerate}
    \item $d_{min} \leftarrow \infty$
    \item $Q_i \leftarrow \text{vide}$
    \item $Q_j \leftarrow \text{vide}$
    \item Pour chaque point $u$ appartenant à $L$ faire :
    \item \quad Pour chaque point $v$ appartenant à $c$ faire :
    \item \quad \quad Si $D[u, v] < d_{min}$ alors :
    \item \quad \quad \quad $d_{min} \leftarrow D[u, v]$
    \item \quad \quad \quad $Q_i \leftarrow u$
    \item \quad \quad \quad $Q_j \leftarrow v$
    \item Retourner $Q_i, Q_j$
\end{enumerate}
\end{tcolorbox}

\subsection{Point le Plus Proche Optimisé (OptPPP)}

\begin{tcolorbox}[colback=darkWhite!75!white,colframe=black!30,left=5pt,right=5pt,top=5pt,bottom=5pt]
\textbf{Algorithme : Procédure Décroise} \\
\textbf{Description :} Améliore un cycle en décroisant les arêtes tant que cela réduit la distance totale. \\
\textbf{Vars :}
\begin{itemize}
    \item $n$ : Entier (Nombre de points)
    \item $\text{amélioration}$ : Booléen (Indicateur de modification)
    \item $i, j$ : Entiers (Indices de boucle)
    \item $u, v, x, y$ : Entiers (Indices des points des arêtes)
    \item $d_{actuelle}, d_{nouvelle}$ : Réels (Distances comparées)
    \item $p, q$ : Entiers (Indices pour l'inversion)
\end{itemize}
\textbf{Entrées :}
\begin{itemize}
    \item $c$ : Cycle hamiltonien initial
    \item $D$ : Matrice des distances
\end{itemize}
\textbf{Sorties :} $c$ (cycle optimisé), $l$ (longueur) \\
\textbf{Début :}
\begin{enumerate}
    \item $n \leftarrow \text{taille}(c)$
    \item $\text{amélioration} \leftarrow \text{Vrai}$
    \item Tant que $\text{amélioration}$ faire :
    \item \quad $\text{amélioration} \leftarrow \text{Faux}$
    \item \quad Pour $i$ allant de $1$ à $n-2$ faire :
    \item \quad \quad Pour $j$ allant de $i+2$ à $n$ faire :
    \item \quad \quad \quad $u \leftarrow c[i]$
    \item \quad \quad \quad $v \leftarrow c[i+1]$
    \item \quad \quad \quad $x \leftarrow c[j]$
    \item \quad \quad \quad Si $j = n$ alors :
    \item \quad \quad \quad \quad $y \leftarrow c[1]$
    \item \quad \quad \quad Sinon :
    \item \quad \quad \quad \quad $y \leftarrow c[j+1]$
    \item \quad \quad \quad $d_{actuelle} \leftarrow D[u, v] + D[x, y]$
    \item \quad \quad \quad $d_{nouvelle} \leftarrow D[u, x] + D[v, y]$
    \item \quad \quad \quad Si $d_{nouvelle} < d_{actuelle}$ alors :
    \item \quad \quad \quad \quad $c[i+1 \dots j] \leftarrow \text{inverser}(c[i+1 \dots j])$
    \item \quad \quad \quad \quad $\text{amélioration} \leftarrow \text{Vrai}$
    \item $l \leftarrow \text{CalculerLongueur}(c, D)$
    \item Retourner $c, l$
\end{enumerate}
\end{tcolorbox}

\vspace{5mm}

\begin{tcolorbox}[colback=darkWhite!75!white,colframe=black!30,left=5pt,right=5pt,top=5pt,bottom=5pt]
\textbf{Algorithme : Procédure OptPPP} \\
\textbf{Description :} Wrapper pour l'algorithme de décroisement. \\
\textbf{Vars :} Aucune \\
\textbf{Entrées :}
\begin{itemize}
    \item $c$ : Cycle hamiltonien initial
    \item $D$ : Matrice des distances
\end{itemize}
\textbf{Sorties :} $c$ (cycle optimisé), $l$ (longueur) \\
\textbf{Début :}
\begin{enumerate}
    \item Retourner $\texttt{Décroise}(c, D)$
\end{enumerate}
\end{tcolorbox}

\subsection{Parcour en largeur d'un ACM à partir de Prim (OptPrim)}


\begin{tcolorbox}[colback=darkWhite!75!white,colframe=black!30,left=5pt,right=5pt,top=5pt,bottom=5pt]
\textbf{Structure de données : GraphePrim}

\textbf{Type} GraphePrim = \textbf{Enregistrement} \\
\quad cle : Tableau[1..n] de Réel; \\
\quad pred : Tableau[1..n] d'Entier; \\
\quad NoeudTas : Tableau[1..n] d'Entier \\
\quad Tas : Tableau[1..n] d'Entier \\1
\textbf{Fin};
\end{tcolorbox}


\begin{tcolorbox}[colback=darkWhite!75!white,colframe=black!30,left=5pt,right=5pt,top=5pt,bottom=5pt]
\textbf{Algorithme : Prim (Optimisé)} \\
\textbf{Description :} Construit un arbre couvrant de poids minimum (MST) en utilisant un tas binaire pour sélectionner efficacement le prochain sommet le plus proche. \\
\textbf{Vars :}
\begin{itemize}
    \item $GraphePrim$ : Structure GraphePrim (Données du graphe et du tas)
    \item $pres$ : Tableau de booléens (Indique si un sommet est dans le tas)
    \item $u, v$ : Entiers (Indices de sommets)
\end{itemize}
\textbf{Entrées :}
\begin{itemize}
    \item $D$ : Matrice des distances $n \times n$
    \item $r0$ : Sommet de départ (par défaut 1)
\end{itemize}
\textbf{Sorties :}
\begin{itemize}
    \item $pred$ : Liste des prédécesseurs définissant l'arbre couvrant
\end{itemize}
\textbf{Début :}
\begin{enumerate}
    \item Initialiser $GraphePrim$ avec $n$ sommets
    \item $pres \leftarrow$ Tableau de booléens initialisé à \texttt{Vrai} (indique si un sommet est dans le tas)
    \item $GraphePrim.cle[r0] \leftarrow 0$
    \item $GraphePrim.echanger(1, GraphePrim.noeudTas[r0])$ \textit{// Placer r0 à la racine}
    \item Tant que $GraphePrim.tas$ n'est pas vide faire :
    \item \quad $u \leftarrow GraphePrim.tas[1]$ \textit{// Extraire le min}
    \item \quad $pres[u] \leftarrow \texttt{Faux}$
    \item \quad $GraphePrim.echanger(1, \text{taille}(tas))$
    \item \quad Retirer le dernier élément de $GraphePrim.tas$
    \item \quad Si $GraphePrim.tas$ non vide alors :
    \item \quad \quad $GraphePrim.versLeBas(1)$
    \item \quad Pour chaque sommet $v$ de $1$ à $n$ faire :
    \item \quad \quad Si $pres[v]$ et $D[u, v] < GraphePrim.cle[v]$ alors :
    \item \quad \quad \quad $GraphePrim.cle[v] \leftarrow D[u, v]$
    \item \quad \quad \quad $GraphePrim.pred[v] \leftarrow u$
    \item \quad \quad \quad $GraphePrim.versLeHaut(GraphePrim.noeudTas[v])$
    \item Retourner $GraphePrim.pred$
\end{enumerate}
\end{tcolorbox}

\vspace{5mm}

\begin{tcolorbox}[colback=darkWhite!75!white,colframe=black!30,left=5pt,right=5pt,top=5pt,bottom=5pt]
\textbf{Algorithme : PredToTree} \\
\textbf{Description :} Convertit la liste des prédécesseurs en une structure d'arbre (dictionnaire d'adjacence). \\
\textbf{Vars :}
\begin{itemize}
    \item $d$ : Dictionnaire (Liste d'adjacence)
    \item $racine$ : Entier (Indice de la racine)
    \item $i$ : Entier (Indice de boucle)
\end{itemize}
\textbf{Entrées :}
\begin{itemize}
    \item $pred$ : Liste des prédécesseurs
\end{itemize}
\textbf{Sorties :} $d$ (arbre), $racine$ \\
\textbf{Début :}
\begin{enumerate}
    \item $d \leftarrow$ Dictionnaire vide
    \item $racine \leftarrow -1$
    \item Pour $i$ allant de $1$ à $\text{taille}(pred)$ faire :
    \item \quad $d[i] \leftarrow$ Liste vide
    \item \quad Si $pred[i]$ est \texttt{None} ou $-1$ alors :
    \item \quad \quad $racine \leftarrow i$
    \item Pour $i$ allant de $1$ à $\text{taille}(pred)$ faire :
    \item \quad Si $pred[i]$ n'est pas \texttt{None} et $pred[i] \neq -1$ alors :
    \item \quad \quad Ajouter $i$ à $d[pred[i]]$
    \item Retourner $d, racine$
\end{enumerate}
\end{tcolorbox}

\vspace{5mm}

\begin{tcolorbox}[colback=darkWhite!75!white,colframe=black!30,left=5pt,right=5pt,top=5pt,bottom=5pt]
\textbf{Algorithme : ParcoursPrefixe} \\
\textbf{Description :} Effectue un parcours en profondeur (DFS) pour obtenir l'ordre de visite des sommets. \\
\textbf{Vars :}
\begin{itemize}
    \item $stack$ : Pile d'entiers
    \item $p$ : Tableau d'entiers (Marquage visite)
    \item $p\_etoile$ : Liste d'entiers (Résultat)
    \item $count$ : Entier
    \item $lst\_voisin$ : Liste d'entiers
    \item $rep$ : Booléen
    \item $voisin$ : Entier
\end{itemize}
\textbf{Entrées :}
\begin{itemize}
    \item $d$ : Dictionnaire d'adjacence
    \item $racine$ : Entier
\end{itemize}
\textbf{Sorties :} $p\_etoile$ (liste ordonnée) \\
\textbf{Début :}
\begin{enumerate}
    \item $stack \leftarrow [racine]$
    \item $p \leftarrow$ Tableau de zéros de taille $n$
    \item $p\_etoile \leftarrow [racine]$
    \item $p[racine] \leftarrow 1$
    \item $count \leftarrow 2$
    \item Tant que $stack$ n'est pas vide faire :
    \item \quad $lst\_voisin \leftarrow d[\text{sommet}(stack)]$
    \item \quad $rep \leftarrow \text{Faux}$
    \item \quad Pour chaque $voisin$ dans $lst\_voisin$ faire :
    \item \quad \quad Si $p[voisin] == 0$ alors :
    \item \quad \quad \quad Empiler $voisin$ sur $stack$
    \item \quad \quad \quad Ajouter $voisin$ à $p\_etoile$
    \item \quad \quad \quad $p[voisin] \leftarrow count$
    \item \quad \quad \quad $count \leftarrow count + 1$
    \item \quad \quad \quad $rep \leftarrow \text{Vrai}$
    \item \quad \quad \quad Sortir de la boucle Pour
    \item \quad Si non $rep$ alors :
    \item \quad \quad Dépiler de $stack$
    \item Retourner $p\_etoile$
\end{enumerate}
\end{tcolorbox}

\vspace{5mm}

\begin{tcolorbox}[colback=darkWhite!75!white,colframe=black!30,left=5pt,right=5pt,top=5pt,bottom=5pt]
\textbf{Algorithme : Procédure OptPrim} \\
\textbf{Description :} Construit un cycle hamiltonien à partir de l'arbre couvrant minimal. \\
\textbf{Vars :}
\begin{itemize}
    \item $pred$ : Liste d'entiers
    \item $d$ : Dictionnaire
    \item $racine$ : Entier
    \item $c$ : Liste d'entiers
    \item $l$ : Réel
\end{itemize}
\textbf{Entrées :}
\begin{itemize}
    \item $points$ : Liste de points
    \item $D$ : Matrice des distances
    \item $r0$ : Sommet de départ
\end{itemize}
\textbf{Sorties :} $c, l$ \\
\textbf{Début :}
\begin{enumerate}
    \item Si $D$ est \texttt{None} alors :
    \item \quad $D \leftarrow \texttt{CalculerMatriceDistances}(points)$
    \item $pred \leftarrow \texttt{Prim}(D, r0)$
    \item $(d, racine) \leftarrow \texttt{PredToTree}(pred)$
    \item $c \leftarrow \texttt{ParcoursPrefixe}(d, racine)$
    \item $l \leftarrow \texttt{CalculerLongueur}(c, D)$
    \item Retourner $c, l$
\end{enumerate}
\end{tcolorbox}

\subsection{OptPrim + Décroisé (OptOptPrim)}

\begin{tcolorbox}[colback=darkWhite!75!white,colframe=black!30,left=5pt,right=5pt,top=5pt,bottom=5pt]
\textbf{Algorithme : Procédure OptOptPrim} \\
\textbf{Description :} Combine OptPrim et le décroisement. \\
\textbf{Vars :}
\begin{itemize}
    \item $c_{init}$ : Liste d'entiers (Cycle initial)
\end{itemize}
\textbf{Entrées :}
\begin{itemize}
    \item $points$ : Liste de points
    \item $D$ : Matrice des distances
    \item $r0$ : Sommet de départ
\end{itemize}
\textbf{Sorties :} $c, l$ \\
\textbf{Début :}
\begin{enumerate}
    \item Si $D$ est \texttt{None} alors :
    \item \quad $D \leftarrow \texttt{CalculerMatriceDistances}(points)$
    \item $(c_{init}, \_) \leftarrow \texttt{OptPrim}(points, D, r0)$
    \item Retourner $\texttt{Décroise}(c_{init}, D)$
\end{enumerate}
\end{tcolorbox}
